last_name / first_name / middle_nameTEXT, NOT NULL

FKdepartment_id → departments.idON DELETE RESTRICT

phoneTEXT NULL

statusTEXT, CHECK (4 значения)

failed_attempts / locked_until
created_atTIMESTAMPTZ
